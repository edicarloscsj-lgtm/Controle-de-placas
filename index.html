Liberar Permissão de Gravação no Supabase

O Supabase retornou o aviso: .
Para liberar 100% a gravação pública (sem bloqueio de RLS) e garantir que todas as colunas existam, copie o comando abaixo, cole no SQL Editor do Supabase e clique em Run: 

-- 1. Cria as tabelas caso não existam
CREATE TABLE IF NOT EXISTS public.placas (
  codigo TEXT PRIMARY KEY,
  status TEXT DEFAULT 'Disponível',
  empresa TEXT,
  responsavel TEXT,
  telefone TEXT,
  cidade TEXT,
  linkgoogle TEXT,
  datavenda TEXT,
  dataativacao TEXT,
  obs TEXT
);

CREATE TABLE IF NOT EXISTS public.historico (
  id SERIAL PRIMARY KEY,
  data TEXT,
  codigo TEXT,
  acao TEXT,
  detalhes TEXT
);

-- 2. Garante que todas as colunas existem (caso a tabela já existisse antes)
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS status TEXT DEFAULT 'Disponível';
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS empresa TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS responsavel TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS telefone TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS cidade TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS linkgoogle TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS datavenda TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS dataativacao TEXT;
ALTER TABLE public.placas ADD COLUMN IF NOT EXISTS obs TEXT;

-- 3. Libera geral o acesso para leitura e gravação (Desativa RLS e cria Política)
ALTER TABLE public.placas DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.historico DISABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Acesso Total Placas" ON public.placas;
CREATE POLICY "Acesso Total Placas" ON public.placas FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Acesso Total Historico" ON public.historico;
CREATE POLICY "Acesso Total Historico" ON public.historico FOR ALL USING (true) WITH CHECK (true);

GRANT ALL ON TABLE public.placas TO anon, authenticated, service_role;
GRANT ALL ON TABLE public.historico TO anon, authenticated, service_role;
